cmake_minimum_required(VERSION 3.15)
project(EtherCATDeviceSimulator LANGUAGES C CXX)
find_package(Threads REQUIRED)
set(BUILD_MASTER_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SLAVE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_SIMULATION ON CACHE BOOL "" FORCE)
set(ENABLE_ESI_PARSER ON CACHE BOOL "" FORCE)
set(KICKCAT_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/KickCAT)
add_executable(cmmt_simulator src/ethercat_slave/main.cc src/ethercat_slave/ipc/sdo_gateway.cc)
target_link_options(cmmt_simulator PRIVATE "-Wl,--wrap=_ZN7kickcat7mailbox8response16createSDOMessageEPNS1_7MailboxEOSt6vectorIhSaIhEE")
target_compile_features(cmmt_simulator PRIVATE cxx_std_17)
target_compile_options(cmmt_simulator PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(cmmt_simulator PRIVATE kickcat_simulation Threads::Threads)
enable_testing()
add_executable(test_fixed_mapping tests/integration/test_fixed_mapping.cc)
target_include_directories(test_fixed_mapping PRIVATE src/ethercat_slave/process_data)
target_compile_features(test_fixed_mapping PRIVATE cxx_std_17)
add_test(NAME fixed_mapping_contract COMMAND test_fixed_mapping)
add_executable(test_multi_slave tests/integration/test_multi_slave.cc)
target_include_directories(test_multi_slave PRIVATE src/ethercat_slave/process_data)
target_compile_features(test_multi_slave PRIVATE cxx_std_17)
target_link_libraries(test_multi_slave PRIVATE kickcat_simulation)
add_test(NAME multi_slave_routing COMMAND test_multi_slave)

add_executable(test_sdo_gateway tests/integration/test_sdo_gateway.cc src/ethercat_slave/ipc/sdo_gateway.cc)
target_include_directories(test_sdo_gateway PRIVATE src/ethercat_slave)
target_compile_features(test_sdo_gateway PRIVATE cxx_std_17)
target_link_libraries(test_sdo_gateway PRIVATE kickcat_simulation Threads::Threads)
target_link_options(test_sdo_gateway PRIVATE "-Wl,--wrap=_ZN7kickcat7mailbox8response16createSDOMessageEPNS1_7MailboxEOSt6vectorIhSaIhEE")

add_executable(test_async_fault tests/integration/test_async_fault.cc)
target_include_directories(test_async_fault PRIVATE src/ethercat_slave)
target_compile_features(test_async_fault PRIVATE cxx_std_17)
target_link_libraries(test_async_fault PRIVATE kickcat_simulation Threads::Threads)
add_test(NAME asynchronous_host_fault COMMAND test_async_fault)
