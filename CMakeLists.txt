cmake_minimum_required(VERSION 3.15)
project(EtherCATDeviceSimulator LANGUAGES C CXX)
set(BUILD_MASTER_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SLAVE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_SIMULATION ON CACHE BOOL "" FORCE)
set(ENABLE_ESI_PARSER ON CACHE BOOL "" FORCE)
set(KICKCAT_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/KickCAT)
add_executable(cmmt_simulator simulator/cmmt_simulator.cc)
target_compile_features(cmmt_simulator PRIVATE cxx_std_17)
target_compile_options(cmmt_simulator PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(cmmt_simulator PRIVATE kickcat_simulation)
enable_testing()
add_executable(test_fixed_mapping tests/test_fixed_mapping.cc)
target_include_directories(test_fixed_mapping PRIVATE simulator)
target_compile_features(test_fixed_mapping PRIVATE cxx_std_17)
add_test(NAME fixed_mapping_contract COMMAND test_fixed_mapping)
add_executable(test_multi_slave tests/test_multi_slave.cc)
target_include_directories(test_multi_slave PRIVATE simulator)
target_compile_features(test_multi_slave PRIVATE cxx_std_17)
target_link_libraries(test_multi_slave PRIVATE kickcat_simulation)
add_test(NAME multi_slave_routing COMMAND test_multi_slave)
